set -Eeuo pipefail

BASE_URL="http://localhost:8000"
REPORT_FILE="shiftwise-ai-llm.md"

curl -v  "${BASE_URL}/health"

curl  --fail-with-body --silent --show-error \
  --request GET \
  --header "Accept: text/markdown" \
  "${BASE_URL}/report/${REPORT_FILE}"

curl  --fail-with-body \
  -X POST "${BASE_URL}/report/${REPORT_FILE}" \
  -H "Content-Type: text/markdown; charset=utf-8" \
  --data-binary "Enumerating objects: 9, done.
Counting objects: 100% (9/9), done.
Delta compression using up to 16 threads
Compressing objects: 100% (4/4), done.
Writing objects: 100% (5/5), 1.04 KiB | 1.04 MiB/s, done.
Total 5 (delta 3), reused 2 (delta 0), pack-reused 0 (from 0)
remote: Resolving deltas: 100% (3/3), completed with 3 local objects.
To github.com:ShiftWise-AI/kubeoptix-reporter.git
   e0c7ad6..d4d3446  feature/save -> feature/save"

curl  --fail-with-body "${BASE_URL}/report/${REPORT_FILE}"

PDF_OUTPUT="/home/parraes/Downloads/${REPORT_FILE%.md}.pdf"
PDF_HEADERS="$(mktemp)"
trap 'rm -f "${PDF_HEADERS}"' EXIT

curl --fail-with-body --silent --show-error \
  --request GET \
  --header "Accept: application/pdf" \
  --dump-header "${PDF_HEADERS}" \
  --output "${PDF_OUTPUT}" \
  "${BASE_URL}/report/${REPORT_FILE}/pdf"

grep -Eiq '^content-type:[[:space:]]*application/pdf([;[:space:]]|$)' "${PDF_HEADERS}"
head -c 5 "${PDF_OUTPUT}" | grep -q '^%PDF'
test -s "${PDF_OUTPUT}"

printf 'PDF válido: %s (%s bytes)\n' "${PDF_OUTPUT}" "$(wc -c < "${PDF_OUTPUT}")"
