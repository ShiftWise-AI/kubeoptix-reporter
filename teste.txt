# Application base URL
BASE_URL="https://kubeoptix-reporter-shiftwise-ai.apps-crc.testing"

# Check application health
curl --fail-with-body --silent --show-error -k \
  "${BASE_URL}/health"
printf '\n'

# Stream a Markdown report from /app/data/reports
REPORT_FILE="default-local.md"
curl --fail-with-body --silent --show-error --get -k \
  --data-urlencode "filename=${REPORT_FILE}" \
  "${BASE_URL}/report"
printf '\n'

# Reject a filename without the .md extension (HTTP 400)
curl --silent --show-error --get -k \
  --data-urlencode "filename=teste.txt" \
  --write-out "\nHTTP %{http_code}\n" \
  "${BASE_URL}/report"

# Reject path traversal (HTTP 400)
curl --silent --show-error --get -k \
  --data-urlencode "filename=../default-local.md" \
  --write-out "\nHTTP %{http_code}\n" \
  "${BASE_URL}/report"

# Report a missing Markdown file (HTTP 404)
curl --silent --show-error --get -k \
  --data-urlencode "filename=inexistente.md" \
  --write-out "\nHTTP %{http_code}\n" \
  "${BASE_URL}/report"

# OpenAPI specification and interactive documentation
curl --fail-with-body --silent --show-error -k \
  "${BASE_URL}/openapi.json"
